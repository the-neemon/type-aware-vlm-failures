# P0.2: pinned environment for the Ada runs.
#
# These are EXACT pins, not floors. Activations shift silently across library
# versions, and this project's entire claim rests on activations, so a
# teammate resolving a different transformers build is a correctness bug and
# not a convenience issue. Resolved and verified on Ada, 2 September 2026,
# Python 3.10.12 (Ubuntu 22.04 on the u22 compute nodes).
#
# Build the environment with:
#     bash scripts/setup_env.sh          # run inside an Ada job, not the head node
#
# The head node runs an older glibc than the compute nodes, so a venv built
# there will not import on a gnode. Build it from inside a job.
#
# torch is the cu126 build. The RTX 2080 Ti is Turing (sm_75), which cu126
# wheels still include. Do not "upgrade" to a wheel that has dropped sm_75
# without checking, because the failure mode is a runtime
# "no kernel image is available for execution on the device", not a warning.
--extra-index-url https://download.pytorch.org/whl/cu126

torch==2.8.0+cu126
torchvision==0.23.0+cu126

# transformers is deliberately the last of the 4.x line rather than 5.x.
# Qwen2.5-VL and qwen-vl-utils were written against the 4.x processor and
# generation APIs, and 5.x reworked both. For a project that caches internal
# state, staying on the branch the model shipped against is worth more than
# being current. Revisit only with a deliberate re-cache of everything.
transformers==4.57.6
qwen-vl-utils==0.0.14
accelerate==1.14.0
